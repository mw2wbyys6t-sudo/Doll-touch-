async speak(text: string) {
  const isJapanese = /[\u3040-\u309F\u30A0-\u30FF]/.test(text);
  
  if (isJapanese) {
    // 提取日语部分
    const jpParts = extractJapaneseParts(text);

  } else {

    await this.tts.speak(text, { language: 'zh-CN' });
  }
}

## 六、技术实现建议

### 6.1 当前APP声音架构

```
【声音输出流程】

用户触发 → AI回复生成 → 文本处理 
  ↓
检测语言 → [中文] → 华为TTS少女声
  ↓
[日语] → 罗马音转换 → 华为TTS英语模式
  ↓
播放音频 + 显示原文
```

### 6.2 优化后的TextToSpeechManager

```typescript
// 针对爱莉优化的TTS管理器

export class AiriVoiceManager {
  // 声音参数配置
  private static readonly VOICE_PARAMS = {
    baseSpeed: 1.0,
    basePitch: 1.0,
    baseVolume: 1.5
  };
  
  // 根据情感调整参数
  static getParamsForMood(mood: string): SpeakOptions {
    const moodMap = {
      'happy': { speed: 1.2, pitch: 1.2, volume: 1.6 },
      'gentle': { speed: 0.9, pitch: 1.0, volume: 1.3 },
      'energetic': { speed: 1.15, pitch: 1.2, volume: 1.6 },
      'coquettish': { speed: 1.1, pitch: 1.15, volume: 1.4 }
    };
    
    return moodMap[mood] || this.VOICE_PARAMS;
  }
  
  // 日语处理
  static async speak(text: string, mood: string = 'default') {
    const isJapanese = this.containsJapanese(text);
    
    if (isJapanese) {
      const romaji = this.toRomaji(text);
      const params = this.getParamsForMood(mood);
      await TextToSpeechManager.speak(romaji, params);
    } else {
      const params = this.getParamsForMood(mood);
      await TextToSpeechManager.speak(text, params);
    }
  }
}
```
